  function scrollToFeedback() {
    var el = appEl.querySelector(".verdict");
    if (!el) return;
    var top = el.getBoundingClientRect().top;
    // 判定が画面の上側にすでに見えている場合（縦
に広いPC画面など）は動かさない
    if (top >= 0 && top <= window.innerHeight * 0
.4) return;
    var reduceMotion =
      window.matchMedia && window.matchMedia("(pr
efers-reduced-motion: reduce)").matches;
    el.scrollIntoView({ block: "start", behavior:
 reduceMotion ? "auto" : "smooth" });
  }
